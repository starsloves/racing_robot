# ═══════════════════════════════════════════════════════════════════════════
# ROS 2 竞速机器人启动命令速查
# ═══════════════════════════════════════════════════════════════════════════

# 环境准备
cd /home/sunrise/dev_ws
source /opt/ros/humble/setup.bash
source install/setup.bash

# ═══════════════════════════════════════════════════════════════════════════
# 一、全局启动（三阶段联动）
# ═══════════════════════════════════════════════════════════════════════════
ros2 launch racing_bringup competition_total.launch.py

# ═══════════════════════════════════════════════════════════════════════════
# 二、分阶段独立测试（官方包，自动发布话题）
# ═══════════════════════════════════════════════════════════════════════════

# Stage1 独立测试（起点 0.50, 0.20, ~10°）
# 流程：前进避障 → 扫码 → 后退 → 地图导航到(2.80,3.10) → 切Phase2
ros2 launch racing_stage1 competition_stage1.launch.py \
    standalone_map_overlay:=true \
    enable_test_publisher:=true \
    test_direction:=clockwise

# Stage2 独立测试 - 顺时针（通道入口 2.80, 3.10, 90°）
ros2 launch racing_stage2 competition_stage2.launch.py \
    standalone_map_overlay:=true \
    enable_test_publisher:=true \
    enable_cmd_relay:=true \
    test_direction:=clockwise

# Stage2 独立测试 - 逆时针
ros2 launch racing_stage2 competition_stage2.launch.py \
    standalone_map_overlay:=true \
    enable_test_publisher:=true \
    enable_cmd_relay:=true \
    test_direction:=counterclockwise

# Stage3 独立测试 - 顺时针（2.38, 3.32, 180°）
ros2 launch racing_stage3 competition_stage3.launch.py \
    standalone_map_overlay:=true \
    enable_test_publisher:=true \
    test_direction:=clockwise

# Stage3 独立测试 - 逆时针（2.38, 3.32, 0°）
ros2 launch racing_stage3 competition_stage3.launch.py \
    standalone_map_overlay:=true \
    enable_test_publisher:=true \
    test_direction:=counterclockwise

# ═══════════════════════════════════════════════════════════════════════════
# 三、工具命令
# ═══════════════════════════════════════════════════════════════════════════

# 紧急停车
ros2 topic pub --once /cmd_vel geometry_msgs/msg/Twist \
    "{linear: {x: 0.0}, angular: {z: 0.0}}"

# 相机启动 + 推理 + Web 展示（All-in-One）
python3 ~/dev_ws/src/racing/tools/camera_all_in_one.py

# 查看话题列表
ros2 topic list

# 查看里程计（验证初始位置）
ros2 topic echo /odom_combined

# 查看阶段和方向话题（验证测试发布器）
ros2 topic echo /competition_phase
ros2 topic echo /competition_qr_task

# 查看 TF 树
ros2 run tf2_tools view_frames

# 实时监听 map→odom TF
ros2 run tf2_ros tf2_echo map odom_combined

# ═══════════════════════════════════════════════════════════════════════════
# 四、编译命令
# ═══════════════════════════════════════════════════════════════════════════

# 编译所有包
colcon build --symlink-install

# 编译官方阶段包
colcon build --symlink-install --packages-select \
    racing_common racing_stage1 racing_stage2 racing_stage3

# 清理重新编译
rm -rf build/ install/ log/
colcon build --symlink-install

# ═══════════════════════════════════════════════════════════════════════════
# 附录：各阶段初始位置参数对照表
# ═══════════════════════════════════════════════════════════════════════════
# 
# 阶段    | map_to_odom_x | map_to_odom_y | map_to_odom_yaw  | 说明
# --------|---------------|---------------|------------------|------------------
# Stage1  | 0.50          | 0.20          | 0.1745 (~10°)    | 起点区域
# Stage2  | 2.80          | 3.10          | 1.5708 (90°)     | 通道入口
# Stage3  | 2.38          | 3.32          | 3.1416 (180°)    | 整圈终点（顺时针）
# Stage3  | 2.38          | 3.32          | 0.0 (0°)         | 整圈终点（逆时针）
#
# 独立测试自动发布话题：
#   /competition_phase (Int32) - 阶段编号 (1/2/3)
#   /competition_qr_task (String) - 方向 (clockwise/counterclockwise)
